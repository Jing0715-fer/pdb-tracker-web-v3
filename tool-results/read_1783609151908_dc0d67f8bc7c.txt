    90→    90→const STORAGE_KEY = 'pdb-tracker:llm-cfg:v1';
    91→    91→const STORAGE_PROVIDER_KEY = 'pdb-tracker:llm-provider:v1';
    92→    92→
    93→    93→/** Sentinel value: the user wants the orchestrator to pick the best provider
    94→    94→ *  based on what the server detected (this is the default behavior). */
    95→    95→const AUTO_PROVIDER = '__auto__';
    96→    96→
    97→    97→function loadStoredProvider(): string {
    98→    98→  if (typeof window === 'undefined') return AUTO_PROVIDER;
    99→    99→  try { return localStorage.getItem(STORAGE_PROVIDER_KEY) || AUTO_PROVIDER; } catch { return AUTO_PROVIDER; }
   100→   100→}
   101→   101→
   102→   102→function loadStoredCfg(): LlmUserConfig {
   103→   103→  if (typeof window === 'undefined') return DEFAULT_LLM_CFG;
   104→   104→  try {
   105→   105→    const raw = localStorage.getItem(STORAGE_KEY);
   106→   106→    if (!raw) return DEFAULT_LLM_CFG;
   107→   107→    return { ...DEFAULT_LLM_CFG, ...JSON.parse(raw) };
   108→   108→  } catch { return DEFAULT_LLM_CFG; }
   109→   109→}
   110→   110→
   111→   111→function persistProvider(p: string) {
   112→   112→  try { localStorage.setItem(STORAGE_PROVIDER_KEY, p); } catch {/* ignore */}
   113→   113→}
   114→   114→
   115→   115→function persistCfg(c: LlmUserConfig) {
   116→   116→  try {
   117→   117→    // Don't persist apiKey in plain localStorage in production — but for
   118→   118→    // this internal tool it's OK. (In a real deploy, use a server-side
   119→   119→    // session and reference it by token here.)
   120→   120→    localStorage.setItem(STORAGE_KEY, JSON.stringify(c));
   121→   121→  } catch {/* ignore */}
   122→   122→}
