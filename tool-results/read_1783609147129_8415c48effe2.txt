    82→const DEFAULT_LLM_CFG: LlmUserConfig = {
    83→  provider: '',
    84→  apiKey: '',
    85→  baseUrl: '',

    90→const STORAGE_KEY = 'pdb-tracker:llm-cfg:v1';
    91→const STORAGE_PROVIDER_KEY = 'pdb-tracker:llm-provider:v1';
    92→
    93→/** Sentinel value: the user wants the orchestrator to pick the best provider
    94→ *  based on what the server detected (this is the default behavior). */
    95→const AUTO_PROVIDER = '__auto__';
    96→
    97→function loadStoredProvider(): string {
    98→  if (typeof window === 'undefined') return AUTO_PROVIDER;
    99→  try { return localStorage.getItem(STORAGE_PROVIDER_KEY) || AUTO_PROVIDER; } catch { return AUTO_PROVIDER; }
   100→}
   101→
   102→function loadStoredCfg(): LlmUserConfig {
   103→  if (typeof window === 'undefined') return DEFAULT_LLM_CFG;
   104→  try {
   105→    const raw = localStorage.getItem(STORAGE_KEY);
   106→    if (!raw) return DEFAULT_LLM_CFG;
   107→    return { ...DEFAULT_LLM_CFG, ...JSON.parse(raw) };
   108→  } catch { return DEFAULT_LLM_CFG; }
   109→}
   110→
   111→function persistProvider(p: string) {
   112→  try { localStorage.setItem(STORAGE_PROVIDER_KEY, p); } catch {/* ignore */}
   113→}
   114→
   115→function persistCfg(c: LlmUserConfig) {
   116→  try {
   117→    // Don't persist apiKey in plain localStorage in production — but for
   118→    // this internal tool it's OK. (In a real deploy, use a server-side
   119→    // session and reference it by token here.)
   120→    localStorage.setItem(STORAGE_KEY, JSON.stringify(c));
   121→  } catch {/* ignore */}
   122→}
